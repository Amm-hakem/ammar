    <script>
        let myChart = null;

        document.getElementById('enableTaxDeduction').addEventListener('change', function() {
            const taxSection = document.getElementById('tax-section');

                taxSection.classList.remove('hidden');

        });

        function resetDefaults() {
            document.getElementById('purchasePrice').value = 1000000;
            document.getElementById('downPaymentPercent').value = 20;
            document.getElementById('interestRate').value = 5.5;
            document.getElementById('loanTerm').value = 25;
            document.getElementById('buyingCosts').value = 2.0;
            document.getElementById('sellingCosts').value = 2.5;
            document.getElementById('hoaFee').value = 500;
            document.getElementById('propertyTaxRate').value = 0.5;
            document.getElementById('maintenanceRate').value = 1.0;
            document.getElementById('appreciationRate').value = 3.0;
            document.getElementById('monthlyRent').value = 4500;
            document.getElementById('rentInflation').value = 3.0;
            document.getElementById('investmentReturn').value = 7.0;
            document.getElementById('enableTaxDeduction').checked = false;
            document.getElementById('tax-section').classList.add('hidden');
            document.getElementById('taxBracket').value = 15;
            calculate();
        }

        function formatCurrency(amount) {
            return new Intl.NumberFormat('ar-SA', { style: 'currency', currency: 'SAR', maximumFractionDigits: 0 }).format(amount);
        }

        function calculate() {
            const purchasePrice = parseFloat(document.getElementById('purchasePrice').value) || 0;
            const downPaymentPct = parseFloat(document.getElementById('downPaymentPercent').value) || 0;
            const annualInterestRate = parseFloat(document.getElementById('interestRate').value) || 0;
            const loanTermYears = parseFloat(document.getElementById('loanTerm').value) || 25;
            const buyingCostsPct = parseFloat(document.getElementById('buyingCosts').value) || 0;
            const sellingCostsPct = parseFloat(document.getElementById('sellingCosts').value) || 0;
            const hoaMonthly = parseFloat(document.getElementById('hoaFee').value) || 0;
            const propertyTaxPct = parseFloat(document.getElementById('propertyTaxRate').value) || 0;
            const maintenancePct = parseFloat(document.getElementById('maintenanceRate').value) || 0;
            const appreciationPct = parseFloat(document.getElementById('appreciationRate').value) || 0;
            
            let currentRent = parseFloat(document.getElementById('monthlyRent').value) || 0;
            const rentInflationPct = parseFloat(document.getElementById('rentInflation').value) || 0;
            const investmentReturnPct = parseFloat(document.getElementById('investmentReturn').value) || 0;

            const enableTax = document.getElementById('enableTaxDeduction').checked;
            const taxBracketPct = enableTax ? (parseFloat(document.getElementById('taxBracket').value) || 0) / 100 : 0;

            const downPayment = purchasePrice * (downPaymentPct / 100);
            const initialBuyingCosts = purchasePrice * (buyingCostsPct / 100);
            const loanAmount = purchasePrice - downPayment;
            
            const monthlyInterestRate = (annualInterestRate / 100) / 12;
            const totalPaymentsCount = loanTermYears * 12;
            let monthlyMortgage = 0;
            if (monthlyInterestRate > 0) {
                monthlyMortgage = loanAmount * (monthlyInterestRate * Math.pow(1 + monthlyInterestRate, totalPaymentsCount)) / (Math.pow(1 + monthlyInterestRate, totalPaymentsCount) - 1);
            } else {
                monthlyMortgage = loanAmount / totalPaymentsCount;
            }

            let renterPortfolio = downPayment + initialBuyingCosts;

            let currentLoanBalance = loanAmount;
            let currentPropertyValue = purchasePrice;
            
            let cumulativeBuyPaid = downPayment + initialBuyingCosts;
            let cumulativeRentPaid = 0;

            let yearlyData = [];
            let chartLabels = ['البداية (السنة 0)'];
            let chartBuyNetWorth = [downPayment];
            let chartRentNetWorth = [renterPortfolio];

            yearlyData.push({
                year: 0,
                propValue: purchasePrice,
                loanBalance: loanAmount,
                equity: downPayment,
                buyTotalPaid: cumulativeBuyPaid,
                rentTotalPaid: 0,
                renterPortfolio: renterPortfolio,
                diff: 0,
                diffWho: 'متساوي'
            });

            for (let year = 1; year <= 10; year++) {
                let yearInterestPaid = 0;
                let yearPrincipalPaid = 0;
                let yearPropertyTax = currentPropertyValue * (propertyTaxPct / 100);
                let yearMaintenance = currentPropertyValue * (maintenancePct / 100);
                let yearHoa = hoaMonthly * 12;

                for (let m = 1; m <= 12; m++) {
                    if (currentLoanBalance > 0) {
                        let interestPayment = currentLoanBalance * monthlyInterestRate;
                        let principalPayment = monthlyMortgage - interestPayment;
                        if (principalPayment > currentLoanBalance) principalPayment = currentLoanBalance;
                        
                        yearInterestPaid += interestPayment;
                        yearPrincipalPaid += principalPayment;
                        currentLoanBalance -= principalPayment;
                    }
                }

                let taxSavings = 0;
                if (enableTax) {
                    let deductibleInterestAndTax = yearInterestPaid + yearPropertyTax;
                    taxSavings = deductibleInterestAndTax * taxBracketPct;
                }

                let annualMortgageTotal = monthlyMortgage * 12;
                let netOwnerCostYear = annualMortgageTotal + yearPropertyTax + yearMaintenance + yearHoa - taxSavings;
                cumulativeBuyPaid += netOwnerCostYear;

                let annualRentTotal = currentRent * 12;
                cumulativeRentPaid += annualRentTotal;

                let ownerOutOfPocketThisYear = netOwnerCostYear;
                if (year === 1) {
                    ownerOutOfPocketThisYear += downPayment + initialBuyingCosts;
                }
                
                renterPortfolio = renterPortfolio * (1 + (investmentReturnPct / 100));
                let cashFlowDiff = ownerOutOfPocketThisYear - annualRentTotal;
                renterPortfolio += cashFlowDiff;

                currentRent *= (1 + (rentInflationPct / 100));
                currentPropertyValue *= (1 + (appreciationRate / 100));

                let sellingCostsAmount = currentPropertyValue * (sellingCostsPct / 100);
                let buyerEquity = currentPropertyValue - currentLoanBalance;
                let buyerNetWorth = buyerEquity - sellingCostsAmount;

                let diff = buyerNetWorth - renterPortfolio;

                chartLabels.push(`السنة ${year}`);
                chartBuyNetWorth.push(Math.round(buyerNetWorth));
                chartRentNetWorth.push(Math.round(renterPortfolio));

                yearlyData.push({
                    year: year,
                    propValue: currentPropertyValue,
                    loanBalance: currentLoanBalance,
                    equity: buyerEquity,
                    buyTotalPaid: cumulativeBuyPaid,
                    rentTotalPaid: cumulativeRentPaid,
                    renterPortfolio: renterPortfolio,
                    diff: Math.abs(diff),
                    diffWho: diff >= 0 ? 'الشراء' : 'الاستئجار',
                    rawDiff: diff
                });
            }

            const finalYear = yearlyData[10];
            const finalBuyNW = chartBuyNetWorth[10];
            const finalRentNW = chartRentNetWorth[10];
            const netDiff = Math.abs(finalBuyNW - finalRentNW);
            const betterOption = finalBuyNW >= finalRentNW ? 'الشراء' : 'الاستئجار';

            document.getElementById('res-prop-value').innerText = formatCurrency(finalYear.propValue);
            document.getElementById('res-loan-balance').innerText = formatCurrency(finalYear.loanBalance);
            document.getElementById('res-equity').innerText = formatCurrency(finalYear.equity);
            document.getElementById('res-buy-total-paid').innerText = formatCurrency(finalYear.buyTotalPaid);
            document.getElementById('res-buy-net-worth').innerText = formatCurrency(finalBuyNW);

            document.getElementById('res-portfolio').innerText = formatCurrency(finalRentNW);
            document.getElementById('res-rent-total-paid').innerText = formatCurrency(finalYear.rentTotalPaid);
            document.getElementById('res-investment-gains').innerText = formatCurrency(finalRentNW - (downPayment + initialBuyingCosts));
            document.getElementById('res-rent-net-worth').innerText = formatCurrency(finalRentNW);

            document.getElementById('net-worth-diff').innerText = formatCurrency(netDiff);

            const badge = document.getElementById('recommendation-badge');
            const title = document.getElementById('recommendation-title');
            const desc = document.getElementById('recommendation-desc');
            const summaryBanner = document.getElementById('summary-banner');

            if (betterOption === 'الشراء') {
                badge.className = "px-3 py-1 rounded-full text-xs font-bold uppercase tracking-wider bg-emerald-100 text-emerald-800 inline-block mb-2";
                badge.innerText = "الخيار الأفضل: شراء العقار";
                title.innerText = "تملك العقار يحقق ثروة أكبر على مدى 10 سنوات";
                desc.innerText = `بناءً على المعطيات المدخلة، فإن شراء العقار ونموه الطبيعي يمنحك صافي ثروة أعلى بمقدار ${formatCurrency(netDiff)} مقارنة بالاستئجار واستثمار الفرق في المحفظة.`;
                summaryBanner.className = "mb-8 bg-gradient-to-br from-blue-50 to-indigo-50 rounded-2xl shadow-xl p-6 border-2 border-blue-200 transition-all duration-300";
            } else {
                badge.className = "px-3 py-1 rounded-full text-xs font-bold uppercase tracking-wider bg-amber-100 text-amber-800 inline-block mb-2";
                badge.innerText = "الخيار الأفضل: الاستئجار والاستثمار";
                title.innerText = "الاستئجار واستثمار الدفعة المقدمة يحقق عائداً أعلى";
                desc.innerText = `بناءً على المعطيات والمدخلات، فإن توجيه الدفعة المقدمة والفرق الشهري لعوائد الاستثمار يوفر لك صافي ثروة أعلى بمقدار ${formatCurrency(netDiff)} مقارنة بالتملك.`;
                summaryBanner.className = "mb-8 bg-gradient-to-br from-emerald-50 to-teal-50 rounded-2xl shadow-xl p-6 border-2 border-emerald-200 transition-all duration-300";
            }

            let tableHTML = '';
            yearlyData.forEach((row) => {
                tableHTML += `
                    <tr class="hover:bg-slate-50 transition-colors">
                        <td class="p-3 font-bold text-slate-800">السنة ${row.year}</td>
                        <td class="p-3 text-slate-600">${formatCurrency(row.propValue)}</td>
                        <td class="p-3 text-rose-600">${formatCurrency(row.loanBalance)}</td>
                        <td class="p-3 text-emerald-600 font-semibold">${formatCurrency(row.equity)}</td>
                        <td class="p-3 text-slate-600">${formatCurrency(row.buyTotalPaid)}</td>
                        <td class="p-3 text-slate-600">${formatCurrency(row.rentTotalPaid)}</td>
                        <td class="p-3 text-indigo-600 font-semibold">${formatCurrency(row.renterPortfolio)}</td>
                        <td class="p-3 font-bold ${row.rawDiff >= 0 ? 'text-blue-600' : 'text-emerald-600'}">
                            ${row.year === 0 ? '-' : `${row.diffWho} (${formatCurrency(row.diff)})`}
                        </td>
                    </tr>
                `;
            });
            document.getElementById('breakdown-table-body').innerHTML = tableHTML;

            updateChart(chartLabels, chartBuyNetWorth, chartRentNetWorth);
        }

        function updateChart(labels, buyData, rentData) {
            const ctx = document.getElementById('netWorthChart').getContext('2d');
            
            if (myChart) {
                myChart.destroy();
            }

            myChart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [
                        {
                            label: 'الشراء (صافي الثروة)',
                            data: buyData,
                            borderColor: '#2563eb',
                            backgroundColor: 'rgba(37, 99, 235, 0.05)',
                            borderWidth: 3,
                            fill: true,
                            tension: 0.3
                        },
                        {
                            label: 'الاستئجار (المحفظة)',
                            data: rentData,
                            borderColor: '#10b981',
                            backgroundColor: 'rgba(16, 185, 129, 0.05)',
                            borderWidth: 3,
                            fill: true,
                            tension: 0.3
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    let label = context.dataset.label || '';
                                    if (label) label += ': ';
                                    if (context.parsed.y !== null) {
                                        label += new Intl.NumberFormat('ar-SA', { style: 'currency', currency: 'SAR', maximumFractionDigits: 0 }).format(context.parsed.y);
                                    }
                                    return label;
                                }
                            }
                        }
                    },
                    scales: {
                        y: {
                            ticks: {
                                font: { family: 'Cairo' },
                                callback: function(value) {
                                    if (value >= 1000000) return (value / 1000000).toFixed(1) + ' م.ر.س';
                                    return (value / 1000).toFixed(0) + ' ألف';
                                }
                            }
                        },
                        x: {
                            ticks: { font: { family: 'Cairo' } }
                        }
                    }
                }
            });
        }
